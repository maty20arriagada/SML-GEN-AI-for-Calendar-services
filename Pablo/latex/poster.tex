% 1. TASK SPEC
\begin{tcolorbox}[colback=white,colframe=agendaAccent,arc=2pt,boxrule=0.7pt,title={\bfseries\small 1. Task Specification \quad {\normalfont\scriptsize(10 pts)}},fonttitle=\color{white},colbacktitle=agendaAccent,left=3pt,right=3pt,top=2pt,bottom=3pt]
\scriptsize
\textbf{Objetivo:} SLM $\leq$8B convierte utterance en \textbf{espanol coloquial} + contexto calendario $\rightarrow$ \textbf{JSON estructurado} con timezone y deteccion de choques. Tarea real, medible, habilita scaffolding futuro.

\textbf{Input:} \texttt{utterance (ES)} + \texttt{now=ISO8601 America/Santiago} + \texttt{calendar\_context=[Event]}\\
\textbf{Output:} \texttt{Array[Event]} schema fijo (RFC5545 subset):\\
\texttt{\{title:str, start:ISO8601(-04:00), end:ISO8601, attendees:[str], location:str|null, rrule:str|null, conflict:bool\}}

\textbf{Correcto} iff 1) JSON valido \& schema OK, 2) \texttt{start/end} exactos con offset, 3) \texttt{conflict=true} iff solapa con \texttt{calendar\_context}, 4) \texttt{rrule} RFC5545 valida.

\vspace{3pt}
\fcolorbox{agendaAccent!30}{agendaLight}{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{
\scriptsize \textbf{Ej. canonico} \hfill {\color{black!60}now=2026-08-31T14:00 (lun) | ctx=[Clase 2026-09-01 08:00--10:00]}\\
\textit{``Agendame reunion con Ana el proximo martes a las 3pm por 1h, avisame si choca''}\\
$\rightarrow$ \texttt{[\{title:``Reunion con Ana'', start:``2026-09-01T15:00:00-04:00'', end:``2026-09-01T16:00:00-04:00'', attendees:[``Ana''], conflict:false\}]}
}}
\vspace{3pt}

\textbf{Metrica:} \texttt{JSON-validity} + \texttt{Exact-Match temporal} + \texttt{Conflict-F1} sobre 20 casos (10 simples + 10 trampas). Gold: \texttt{dataset/examples\_es.json} -- \texttt{America/Santiago -04:00}.
\end{tcolorbox}

% 2. FAILURE DIAGNOSIS
\begin{tcolorbox}[colback=white,colframe=warnRed,arc=2pt,boxrule=0.7pt,title={\bfseries\small 2. Failure Diagnosis \quad {\normalfont\scriptsize(10 pts -- evidencia)}},fonttitle=\color{white},colbacktitle=warnRed,left=3pt,right=3pt,top=2pt,bottom=3pt]
\scriptsize
\textbf{Setup:} prompting directo \textbf{zero-shot}, \texttt{temp=0}, system: ``Devuelve SOLO JSON array...''. Modelo: \textbf{Gemma-2-2B-it} (representante $\leq$8B) en Colab Free T4. Evaluado sobre \textbf{20 utterances ES}.

\begin{center}
\renewcommand{\arraystretch}{1.15}
\begin{tabular}{lcc}
\toprule
\textbf{Metrica (20 casos)} & \textbf{Direct} & \textbf{Target} \\
\midrule
JSON valido & 70\% (14/20) & 100\% \\
Exact-Match temporal & \textbf{45\% (9/20)} & $\geq$85\% \\
Conflict-F1 & 38\% & $\geq$80\% \\
Casos con $\geq$1 error & 75\% & $<$20\% \\
\bottomrule
\end{tabular}
\end{center}

\textbf{5 modos de fallo sistematico:}
\begin{itemize}
\item \textbf{Temporal off-by-week (35\%):} ``proximo martes'' desde lun $\rightarrow$ predice 2026-09-08 en vez de 2026-09-01.
\item \textbf{Formato sin timezone (25\%):} ``3pm'' $\rightarrow$ \texttt{03:00} o \texttt{2026-09-01T15:00} sin \texttt{-04:00}.
\item \textbf{Omite conflicto (40\%):} martes 9am vs clase 8--10 $\rightarrow$ \texttt{conflict=false} (debe ser true).
\item \textbf{Recurrencia rota (30\%):} ``cada martes 7am'' $\rightarrow$ evento unico, \texttt{rrule=null}.
\item \textbf{Multi-evento/ambiguedad:} 2 reuniones o ``a las 8'' $\rightarrow$ 1 evento o 20:00.
\end{itemize}

\fcolorbox{warnRed!40}{warnRed!7}{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{
\scriptsize \textbf{Evidencia (2 fallos reales Gemma-2B zero-shot):}\\
\texttt{ID11:} Esperado \texttt{2026-09-01T15:00-04:00} $\rightarrow$ Predijo \texttt{2026-09-08T15:00-04:00} \colorbox{warnRed!20}{\textbf{+7 dias}} \\
\texttt{ID12:} Esperado \texttt{conflict:true} (9am choca 8--10) $\rightarrow$ Predijo \texttt{conflict:false + JSON sin offset}
}}
\vspace{2pt}

\textit{Conclusion:} fallo no es aleatorio sino de \textbf{razonamiento temporal + constrained decoding}. Justifica D2--D4: date-parser auxiliar + CoT + function calling.
\end{tcolorbox}
